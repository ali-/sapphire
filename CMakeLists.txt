cmake_minimum_required(VERSION 3.20)
project(sapphire LANGUAGES CXX)

if(WIN32)
    set(_sapphire_default_backend directx)
elseif(APPLE)
    set(_sapphire_default_backend metal)
else()
    set(_sapphire_default_backend "")
endif()
set(backend "${_sapphire_default_backend}" CACHE STRING "Sapphire graphics backend")
set_property(CACHE backend PROPERTY STRINGS metal directx)

find_package(SDL3 CONFIG REQUIRED)
add_library(sapphire INTERFACE)
add_library(Sapphire::Sapphire ALIAS sapphire)
target_compile_features(sapphire INTERFACE cxx_std_17)
target_link_libraries(sapphire INTERFACE SDL3::SDL3)

if(backend STREQUAL "metal")
    if(NOT APPLE)
        message(FATAL_ERROR "The Metal backend requires an Apple target")
    endif()
elseif(backend STREQUAL "directx")
    if(NOT WIN32)
        message(FATAL_ERROR "The DirectX backend requires a Windows target")
    endif()
    find_path(SAPPHIRE_DXC_INCLUDE_DIR dxcapi.h REQUIRED)
    find_library(SAPPHIRE_DXC_LIBRARY NAMES dxcompiler REQUIRED)
    target_include_directories(sapphire INTERFACE "${SAPPHIRE_DXC_INCLUDE_DIR}")
    target_link_libraries(sapphire INTERFACE "${SAPPHIRE_DXC_LIBRARY}")
else()
    message(FATAL_ERROR "Select a supported backend with -D backend=metal or -D backend=directx")
endif()

target_compile_definitions(sapphire INTERFACE "SAPPHIRE_BACKEND_HEADER=\"${CMAKE_CURRENT_SOURCE_DIR}/backend/${backend}.h\"")
